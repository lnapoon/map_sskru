#include <WiFi.h>
#include <HTTPClient.h>
#include <SPI.h>
#include <Adafruit_GFX.h>
#include <Adafruit_ILI9341.h>

// กำหนดขา Pin ตามใบงานที่ 4 (Task 2)
#define TFT_CS    5
#define TFT_DC    2
#define TFT_RST   4
#define BUZZER_PIN 25

Adafruit_ILI9341 tft = Adafruit_ILI9341(TFT_CS, TFT_DC, TFT_RST);

// ปีการศึกษาปัจจุบัน (2569 -> ใช้เลข 69 คำนวณชั้นปี)
const int CURRENT_ACADEMIC_YEAR = 69;

// ใส่ URL ของ Google Apps Script เมื่อสร้างเสร็จ (ตอนนี้เว้นไว้ทดสอบใน Wokwi ได้เลย)
const char* GAS_URL = "";

void beepSuccess() {
  tone(BUZZER_PIN, 2500, 150);
}

void beepError() {
  tone(BUZZER_PIN, 600, 400);
}

void showReadyScreen() {
  tft.fillScreen(ILI9341_NAVY);
  tft.setTextColor(ILI9341_CYAN);
  tft.setTextSize(2);
  tft.setCursor(20, 20);
  tft.println("SSKRU LIBRARY");
  tft.setTextColor(ILI9341_WHITE);
  tft.setCursor(20, 50);
  tft.println("CARD - X SCANNER");
  tft.drawFastHLine(20, 80, 200, ILI9341_WHITE);
  tft.setTextSize(2);
  tft.setTextColor(ILI9341_YELLOW);
  tft.setCursor(20, 110);
  tft.println("Ready to Scan...");
  tft.setTextSize(1);
  tft.setTextColor(ILI9341_LIGHTGREY);
  tft.setCursor(20, 145);
  tft.println("Type 10-digit Student ID");
  tft.setCursor(20, 160);
  tft.println("in Serial Monitor below");
}

void setup() {
  Serial.begin(115200);
  pinMode(BUZZER_PIN, OUTPUT);

  tft.begin();
  tft.setRotation(1);
  showReadyScreen();

  // เชื่อมต่อ WiFi จำลองของ Wokwi
  WiFi.begin("Wokwi-GUEST", "", 6);
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(250);
    Serial.print(".");
  }
  Serial.println("\nWiFi Connected!");
  Serial.println("=========================================");
  Serial.println("จำลองการสแกนบัตร: พิมพ์รหัส 10 หลัก เช่น 6812732120 แล้วกด Enter");
  Serial.println("=========================================");
}

void processStudentCard(String code, String entryType) {
  code.trim();

  // ตรวจสอบว่าเป็นตัวเลข 10 หลักหรือไม่
  if (code.length() != 10) {
    beepError();
    Serial.println("[ERROR] รหัสต้องมี 10 หลักเท่านั้น!");
    return;
  }

  // ถอดรหัส 5 ส่วน ตามโครงสร้างรหัสนักศึกษา มรภ.ศรีสะเกษ ของอาจารย์
  String yearPrefix = code.substring(0, 2);   // หลักที่ 1-2: เช่น "68"
  String eduType    = code.substring(2, 3);   // หลักที่ 3:   เช่น "1" (ป.ตรี ปกติ)
  String majorCode  = code.substring(3, 7);   // หลักที่ 4-7: เช่น "2732" (วิทยาการคอมฯ)
  String roomNo     = code.substring(7, 8);   // หลักที่ 8:   เช่น "1" (ห้อง 1)
  String studentNo  = code.substring(8, 10);  // หลักที่ 9-10: เช่น "20" (เลขที่ 20)

  int entryYear = yearPrefix.toInt();
  int yearLevel = (CURRENT_ACADEMIC_YEAR - entryYear) + 1;
  if (yearLevel < 1) yearLevel = 1;

  // สร้าง JSON ตามที่ตกลงในใบงานที่ 4
  String jsonPayload = "{\"student_id\":\"" + code + "\","
                       "\"year_level\":" + String(yearLevel) + ","
                       "\"edu_type\":\"" + eduType + "\","
                       "\"major_code\":\"" + majorCode + "\","
                       "\"room_no\":\"" + roomNo + "\","
                       "\"student_no\":\"" + studentNo + "\","
                       "\"entry_type\":\"" + entryType + "\"}";

  beepSuccess();

  // แสดงผลออกทาง Serial Monitor
  Serial.println("\n--- [SCAN SUCCESS] ---");
  Serial.println("รหัสนักศึกษา : " + code);
  Serial.println("1) ชั้นปีที่    : ปี " + String(yearLevel) + " (รหัส " + yearPrefix + ")");
  Serial.println("2) ระดับ     : " + eduType + " (1=ปกติ ป.ตรี)");
  Serial.println("3) รหัสสาขา  : " + majorCode);
  Serial.println("4) หมู่เรียน   : ห้อง " + roomNo);
  Serial.println("5) เลขที่     : " + studentNo);
  Serial.println("JSON ที่ส่ง   : " + jsonPayload);

  // แสดงผลบนหน้าจอสี TFT LCD
  tft.fillScreen(ILI9341_DARKGREEN);
  tft.setTextColor(ILI9341_WHITE);
  tft.setTextSize(2);
  tft.setCursor(20, 20);
  tft.println("WELCOME TO LIBRARY!");
  tft.drawFastHLine(20, 48, 280, ILI9341_WHITE);

  tft.setTextColor(ILI9341_YELLOW);
  tft.setCursor(20, 65);
  tft.println("ID: " + code);

  tft.setTextColor(ILI9341_WHITE);
  tft.setCursor(20, 100);
  tft.println("Year Level : Year " + String(yearLevel));
  tft.setCursor(20, 130);
  tft.println("Major Code : " + majorCode);
  tft.setCursor(20, 160);
  tft.println("Room: " + roomNo + "  |  No: " + studentNo);

  delay(3000);
  showReadyScreen();
}

void loop() {
  if (Serial.available() > 0) {
    String input = Serial.readStringUntil('\n');
    input.trim();
    if (input.length() > 0) {
      processStudentCard(input, "SCAN");
    }
  }
}
